# ^ for start
# $ for end
# / one or more
# * zero or more
# + or
def match(pattern, pattern2):
    if "+" in pattern:
        parts= pattern.split("+")
        return any(match(p, pattern2) for p in parts)
    if "^" in pattern:
        pattern= pattern[1:]
        if not pattern2.startswith(pattern[0]):
            return False
        
    if "$" in pattern:
            pattern= pattern.replace("$", "")
            if not pattern2.endswith(pattern[-1]):
                return False
            
    i=0
    j=0
    while i < len(pattern) and j < len(pattern2):
            if pattern [i]== "/":
                i += 1
                if i >= len(pattern):
                    return False
                x= pattern[i]
                if pattern2[j] != x:
                    return False
                while j< len(pattern2) and pattern2[j]== x:
                    j +=1
                i+=1
            elif pattern[i]== "*":
                i+=1
                if i>= len(pattern):
                    return False
                x= pattern[i]
                while j< len(pattern2) and pattern2[j]== x:
                    j+=1
                i +=1
            else:
                if pattern[i] != pattern2[j]:
                    return False
                i+=1
                j+=1

    return i >= len(pattern) and j>= len(pattern2)
def main():
     tests= [
          ("^1/0", "1000"),
          ("^1/0$", "1000"),
          ("^1/0", "000"),
          ("^1+0", "1"),
          ("1+3", "0")
     ]
     for pattern, pattern2 in tests:
        if match(pattern, pattern2)== True:
               print('(',pattern,',', pattern2,')', "regular expression matches with output.")
        else:
             print('(',pattern,',', pattern2,')', "regular expression does not match with output.")
main()
